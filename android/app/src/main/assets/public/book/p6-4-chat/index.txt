export default function ChatBox() {
  const [message, setMessage] = useState("");
  const [messages, setMessages] = useState([]);

  useEffect(() => {
    socket.on("receive_message", (data) => {
      setMessages((prev) => [...prev, data]);
    });
    return () => socket.off("receive_message");
  }, []);

  const sendMessage = () => {
    socket.emit("send_message", { text: message, user: "Manoj" });
    setMessages((prev) => [...prev, { text: message, user: "Manoj" }]);
    setMessage("");
  };

  return (
    <div className="p-6 bg-slate-900 border rounded-3xl space-y-4 max-w-md mx-auto">
      <div className="h-60 overflow-y-auto space-y-2 border-b border-slate-800 pb-4">
        {messages.map((m, idx) => (
          <div key={idx} className="text-xs">
            <span className="font-black text-blue-400">{m.user}:</span> {m.text}
          </div>
        ))}
      </div>
      <div className="flex gap-2">
        <input 
          value={message} 
          onChange={(e) => setMessage(e.target.value)}
          className="flex-1 p-2 rounded-xl bg-slate-950 text-white border text-xs"
        />
        <button onClick={sendMessage} className="px-4 py-2 bg-blue-600 rounded-xl text-xs font-bold text-white">
          Send
        </button>
      </div>
    </div>
  );
