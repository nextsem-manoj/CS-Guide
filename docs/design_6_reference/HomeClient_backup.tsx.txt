export default function HomeClient() {
  const [activeIndex, setActiveIndex] = useState(0);
  const [showUpdateModal, setShowUpdateModal] = useState(false);
  const [hasInteracted, setHasInteracted] = useState(false);
  const touchStartY = useRef(0);
  const totalSlides = 5; // Hero + Curriculum Carousel + Features 1 + Features 2 + Footer
  const [isMobile, setIsMobile] = useState(false);
  const [currentUser, setCurrentUser] = useState<{ name: string; email: string; avatar: string; photo?: string } | null>(null);
  
  const { completedChapters } = useProgress();
  const [selectedPhaseIdx, setSelectedPhaseIdx] = useState(0);
  const milestonesRef = useRef<HTMLDivElement>(null);
  const [isAtStart, setIsAtStart] = useState(true);
  const [isAtEnd, setIsAtEnd] = useState(false);

  const scrollMilestonesLeft = () => {
    if (milestonesRef.current) {
      milestonesRef.current.scrollBy({ left: -220, behavior: "smooth" });
    }
  };

  const scrollMilestonesRight = () => {
    if (milestonesRef.current) {
      milestonesRef.current.scrollBy({ left: 220, behavior: "smooth" });
    }
  };

  const getPhaseProgress = (phase: typeof coursePhases[0]) => {
    if (!completedChapters) return 0;
    const phaseChapters = phase.topics.map((t) => t.path);
    const completed = phaseChapters.filter((path) => completedChapters[path]).length;
    return Math.round((completed / phaseChapters.length) * 100);
  };

  // Offline pre-caching states
  const [downloadingOffline, setDownloadingOffline] = useState(false);
  const [downloadProgress, setDownloadProgress] = useState(0);
  const [downloadCurrent, setDownloadCurrent] = useState(0);
  const [downloadTotal, setDownloadTotal] = useState(0);
  const [downloadComplete, setDownloadComplete] = useState(false);
  const [isAlreadyDownloaded, setIsAlreadyDownloaded] = useState(false);

  const { isInstallable, installApp, isInstalled } = usePWA();

  const handleOfflineFeatureClick = async () => {
    if (downloadingOffline || downloadComplete) return;

    // Check if user is offline when trying to download
    if (typeof window !== 'undefined' && !navigator.onLine) {
      alert("You are currently offline. Please connect to the internet to sync and download full course content.");
      return;
    }

    // If app is installable, prompt user to install it on device
    if (isInstallable) {
      try { await installApp(); } catch (e) {}
    }

    // Build the list of all unique Nextsem pages and chapters to cache
    const paths = ["/", "/about", "/contact", "/playground", "/privacy", "/terms"];
    coursePhases.forEach(phase => {
      phase.topics.forEach(topic => {
        if (topic.path && !paths.includes(`/book/${topic.path}`)) {
          paths.push(`/book/${topic.path}`);
        }
      });
    });

    setDownloadingOffline(true);
    setDownloadProgress(0);
    setDownloadTotal(paths.length);
    setDownloadComplete(false);

    let count = 0;
    for (const path of paths) {
      try {
        // Fetch triggers the Service Worker fetch handler to network-fetch and cache the HTML resource
        await fetch(path);
      } catch (err) {
        console.error(`Offline downloader failed to cache route: ${path}`, err);
      }
      count++;
      setDownloadCurrent(count);
      setDownloadProgress(Math.round((count / paths.length) * 100));
      // Give a tiny pause to prevent browser thread choking
      await new Promise(resolve => setTimeout(resolve, 80));
    }

    setDownloadComplete(true);
    setDownloadingOffline(false);
    setIsAlreadyDownloaded(true);
    localStorage.setItem("nextsem_all_downloaded", "true");
    localStorage.setItem("nextsem_last_online_sync", Date.now().toString());

    // Auto-dismiss success screen after 3 seconds
    setTimeout(() => {
      setDownloadComplete(false);
    }, 3000);
  };

  const handleSearchFeatureClick = () => {
    window.dispatchEvent(new Event("open_search"));
  };

  const handleTranslateFeatureClick = () => {
    window.dispatchEvent(new Event("open_translate"));
  };

  useEffect(() => {
    const checkMobile = () => {
      setIsMobile(window.innerWidth < 768);
    };
    checkMobile();
    window.addEventListener("resize", checkMobile);
    
    const checkAuth = () => {
      const saved = localStorage.getItem('nextsem_user');
      if (saved) {
        try { setCurrentUser(JSON.parse(saved)); } catch (e) { console.error(e); }
      } else {
        setCurrentUser(null);
      }
    };
    checkAuth();
    window.addEventListener("nextsem_auth_changed", checkAuth);

    // Check if all content has already been downloaded
    if (localStorage.getItem("nextsem_all_downloaded") === "true") {
      setTimeout(() => setIsAlreadyDownloaded(true), 0);
    }

    // Trigger update modal on load with a slight delay for smooth entry animation
    const lastUpdateSeen = localStorage.getItem("nextsem_update_seen");
    let timer: NodeJS.Timeout;
    if (lastUpdateSeen !== "1.5") {
      timer = setTimeout(() => setShowUpdateModal(true), 1500);
    }

    return () => {
      window.removeEventListener("resize", checkMobile);
      window.removeEventListener("nextsem_auth_changed", checkAuth);
      if (timer) clearTimeout(timer);
    };
  }, []);

  const handleNext = () => {
    setHasInteracted(true);
    setActiveIndex((prev) => Math.min(totalSlides - 1, prev + 1));
  };

  const handlePrev = () => {
    setHasInteracted(true);
    setActiveIndex((prev) => Math.max(0, prev - 1));
  };

  const handleTouchStart = (e: React.TouchEvent) => {
    touchStartY.current = e.touches[0].clientY;
  };

  const handleTouchEnd = (e: React.TouchEvent) => {
    const touchEndY = e.changedTouches[0].clientY;
    const diff = touchStartY.current - touchEndY;
    if (diff > 50) handleNext();
    else if (diff < -50) handlePrev();
  };

  const selectMilestone = (index: number) => {
    setSelectedPhaseIdx(index);
    if (milestonesRef.current) {
      const childNodes = milestonesRef.current.children;
      // Button index maps to (index * 2) due to connector trace elements
      const buttonIdx = index * 2;
      if (childNodes[buttonIdx]) {
        const child = childNodes[buttonIdx] as HTMLElement;
        const targetScrollLeft = child.offsetLeft - (milestonesRef.current.clientWidth - child.offsetWidth) / 2;
        milestonesRef.current.scrollTo({ left: targetScrollLeft, behavior: "smooth" });
      }
    }
  };

  const handleMilestonesScroll = () => {
    if (!milestonesRef.current) return;
    const { scrollLeft, scrollWidth, clientWidth } = milestonesRef.current;
    setIsAtStart(scrollLeft <= 10);
    setIsAtEnd(scrollLeft + clientWidth >= scrollWidth - 10);
  };

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === "ArrowDown") handleNext();
      if (e.key === "ArrowUp") handlePrev();
    };
    
    let wheelTimeout: NodeJS.Timeout | null = null;
    const handleWheel = (e: WheelEvent) => {
      if (wheelTimeout) return;
      if (e.deltaY > 40) {
        handleNext();
        wheelTimeout = setTimeout(() => { wheelTimeout = null; }, 500);
      } else if (e.deltaY < -40) {
        handlePrev();
        wheelTimeout = setTimeout(() => { wheelTimeout = null; }, 500);
      }
    };

    window.addEventListener("keydown", handleKeyDown);
    window.addEventListener("wheel", handleWheel, { passive: false });
    
    return () => {
      window.removeEventListener("keydown", handleKeyDown);
      window.removeEventListener("wheel", handleWheel);
    };
  }, []);

  const handleStartLearning = () => {
    // Disabled OS-level fullscreen request to prevent viewport/navigation resets on mobile
  };

  const getGlowClass = (index: number) => {
    if (index === 0) return "from-blue-500/20 via-indigo-500/10 to-transparent";
    if (index >= 1 && index <= coursePhases.length) {
      const phase = coursePhases[index - 1];
      const id = phase.id;
      if (id.startsWith("P0")) return "from-blue-500/25 via-sky-500/10 to-transparent";
      if (id.startsWith("P1")) return "from-violet-500/25 via-purple-500/10 to-transparent";
      if (id.startsWith("P2")) return "from-pink-500/25 via-rose-500/10 to-transparent";
      if (id.startsWith("P3")) return "from-amber-500/25 via-yellow-500/10 to-transparent";
      if (id.startsWith("P4")) return "from-cyan-500/25 via-teal-500/10 to-transparent";
      if (id.startsWith("P5")) return "from-emerald-500/25 via-teal-500/10 to-transparent";
      if (id.startsWith("P6")) return "from-fuchsia-500/25 via-pink-500/10 to-transparent";
      if (id.startsWith("P7")) return "from-purple-500/25 via-indigo-500/10 to-transparent";
    }
    return "from-slate-500/15 via-slate-600/10 to-transparent";
  };

  return (
    <div 
      className="h-[100dvh] w-full bg-slate-50 dark:bg-[#060c18] text-gray-900 dark:text-gray-100 font-sans selection:bg-blue-200 relative overflow-hidden select-none transition-colors duration-1000"
      onTouchStart={handleTouchStart}
      onTouchEnd={handleTouchEnd}
    >
      {/* Dynamic Ambient Background Glows (Stationary for Parallax) */}
      <div className="absolute inset-0 z-0 pointer-events-none">
        <div className={`absolute -top-[20%] -left-[20%] w-[80vw] h-[80vw] md:w-[50vw] md:h-[50vw] rounded-full blur-[100px] md:blur-[150px] bg-gradient-to-br ${getGlowClass(activeIndex)} transition-all duration-[1000ms] ease-out animate-slow-rotate`} />
        <div className={`absolute -bottom-[20%] -right-[20%] w-[80vw] h-[80vw] md:w-[50vw] md:h-[50vw] rounded-full blur-[100px] md:blur-[150px] bg-gradient-to-tl ${getGlowClass((activeIndex + 2) % totalSlides)} transition-all duration-[1000ms] ease-out opacity-80 animate-slow-rotate`} />
      </div>
      
      {/* Static Thin Line Navigation */}
      <div 
        className="absolute right-3.5 md:right-8 top-1/2 -translate-y-1/2 w-[3.5px] h-[30dvh] rounded-full transition-all duration-[800ms] ease-out z-50 pointer-events-none"
        style={{ 
          backgroundColor: (slideThemes[activeIndex] || slideThemes[0]).rawColor,
          boxShadow: `0 0 12px ${(slideThemes[activeIndex] || slideThemes[0]).rawColor}`,
          opacity: 0.85
        }}
      />

      {/* Scroll Hint Indicator — fades out after first interaction */}
      <div
        className={`absolute bottom-6 left-1/2 -translate-x-1/2 z-50 flex flex-col items-center gap-1 pointer-events-none transition-all duration-700 ${
          hasInteracted ? "opacity-0 translate-y-4" : "opacity-100 translate-y-0"
        }`}
      >
        <span className="text-[10px] font-semibold uppercase tracking-widest text-white/50">Scroll</span>
        <svg
          className="w-5 h-5 text-white/70 animate-bounce"
          fill="none"
          stroke="currentColor"
          strokeWidth={2.5}
          viewBox="0 0 24 24"
        >
          <path strokeLinecap="round" strokeLinejoin="round" d="M19 9l-7 7-7-7" />
        </svg>
      </div>

      {/* Slide Container (translates Y based on activeIndex) */}
      <div 
        className="w-full h-full transition-transform duration-[850ms] [transition-timing-function:cubic-bezier(0.16,1,0.3,1)] flex flex-col will-change-transform transform-gpu"
        style={{ transform: `translate3d(0, -${activeIndex * 100}dvh, 0)` }}
      >
        
        {/* Slide 0: Hero */}
        <div className="h-[100dvh] w-full shrink-0 flex items-center justify-center pt-20 p-4 md:p-12">
          <div className="flex flex-col lg:flex-row items-center justify-between gap-8 lg:gap-12 w-full h-[76dvh] lg:h-auto max-w-6xl bg-gradient-to-br from-blue-50/70 to-white/90 dark:from-slate-800/80 dark:to-[#060c18]/90 backdrop-blur-xl border border-blue-100/50 dark:border-slate-800/50 rounded-3xl p-6 md:p-12 shadow-2xl transition-opacity duration-1000">
            
            {/* Left Panel: Purpose & CTA */}
            <div className="flex flex-col items-center lg:items-start text-center lg:text-left lg:w-3/5 h-full justify-between lg:justify-start">
              

              {/* Trust badge row */}
              <div className="flex flex-wrap items-center gap-2 mb-4 justify-center lg:justify-start">
                <span className="inline-flex items-center gap-1 text-[10px] font-black uppercase tracking-widest px-2.5 py-1 rounded-full bg-emerald-500/10 text-emerald-500 border border-emerald-500/20">
                  <span style={{ animation: 'hand-bounce 0.9s ease-in-out infinite alternate' }}>🟢</span> Free to Start
                </span>
                <span className="inline-flex items-center gap-1 text-[10px] font-black uppercase tracking-widest px-2.5 py-1 rounded-full bg-amber-500/10 text-amber-500 border border-amber-500/20">
                  <span>🏆</span> Certification Included
                </span>
                <span className="inline-flex items-center gap-1 text-[10px] font-black uppercase tracking-widest px-2.5 py-1 rounded-full bg-blue-500/10 text-blue-500 border border-blue-500/20">
                  <span>📱</span> Works Offline
                </span>
              </div>

              <h2 className="text-xl sm:text-3xl lg:text-5xl font-black text-slate-900 dark:text-white mb-3 md:mb-5 tracking-tight leading-tight uppercase">
                From <span className="relative inline-block">
                  <span className="text-transparent bg-clip-text bg-gradient-to-r from-amber-400 to-orange-500">Zero</span>
                </span> to{" "}
                <span className="text-transparent bg-clip-text bg-gradient-to-r from-blue-500 via-violet-500 to-emerald-400">
                  Job-Ready
                </span>{" "}
                Developer
              </h2>

              {/* Animated tagline */}
              <div className="mb-6 md:mb-8 max-w-xl space-y-3">
                <p className="text-slate-700 dark:text-slate-300 text-sm sm:text-base lg:text-lg font-medium leading-relaxed">
                  <span className="text-lg mr-1" style={{ display: 'inline-block', animation: 'hand-bounce 1s ease-in-out infinite alternate' }}>💡</span>
                  <strong className="text-slate-900 dark:text-white">No fluff. No passive watching.</strong> Just real code, real projects, and real results — structured from <em>absolute beginner</em> to a <em>hireable developer</em>.
                </p>

                {/* Outcome stat pills */}
                <div className="flex flex-wrap gap-2 mt-3">
                  {[
                    { icon: "⚡", label: "Structured in 8 Phases", color: "from-blue-500/15 to-indigo-500/10 border-blue-500/20 text-blue-400" },
                    { icon: "🎯", label: "Hands-on Every Lesson", color: "from-emerald-500/15 to-teal-500/10 border-emerald-500/20 text-emerald-400" },
                    { icon: "🧠", label: "Built for BCA / B.Tech", color: "from-violet-500/15 to-purple-500/10 border-violet-500/20 text-violet-400" },
                  ].map((stat) => (
                    <span
                      key={stat.label}
                      className={`inline-flex items-center gap-1.5 text-[10px] sm:text-xs font-bold px-3 py-1.5 rounded-full bg-gradient-to-r border ${stat.color} dark:${stat.color} backdrop-blur-sm`}
                    >
                      <span>{stat.icon}</span> {stat.label}
                    </span>
                  ))}
                </div>
              </div>
              

              <div className="flex flex-col sm:flex-row gap-3 items-center justify-center lg:justify-start w-full max-w-md mb-2">
                <Link 
                  href="/book/p0-1-setup" 
                  onClick={handleStartLearning}
                  className="w-full sm:w-auto text-black font-black text-xs md:text-sm uppercase tracking-widest py-3.5 px-8 md:py-4 md:px-10 rounded-2xl transition-all transform hover:-translate-y-0.5 flex items-center justify-center gap-2 active:scale-95 shrink-0"
                  style={{
                    background: "#39FF14",
                    boxShadow: "0 0 25px rgba(57,255,20,0.55)",
                  }}
                >
                  <span>🚀</span> Start Learning Now
                </Link>

                {/* Continue with Google Button */}
                <button
                  onClick={async () => {
                    if (currentUser) {
                      try {
                        await signOut(auth);
                      } catch (err) {
                        console.error("Sign Out Error:", err);
                      }
                    } else {
                      try {
                        await signInWithGoogle();
                      } catch (err) {
                        console.error("Sign In Error:", err);
                        const authErr = err as { code?: string };
                        window.dispatchEvent(new CustomEvent("nextsem_auth_error", { detail: authErr.code }));
                      }
                    }
                  }}
                  className="w-full sm:w-auto flex items-center justify-center gap-2.5 px-6 py-3 bg-white dark:bg-slate-900 text-slate-700 dark:text-slate-200 border border-slate-200 dark:border-slate-800 rounded-full font-bold text-xs md:text-sm shadow-md hover:bg-gray-50 dark:hover:bg-slate-800 hover:border-slate-300 dark:hover:border-slate-700 hover:-translate-y-0.5 transition-all active:scale-95 shrink-0"
                >
                  {currentUser ? (
                    <>
                      {currentUser.photo ? (
                        <img src={currentUser.photo} alt="" className="w-5 h-5 rounded-full shrink-0 object-cover border border-white/20" />
                      ) : (
                        <div className="w-5 h-5 rounded-full bg-blue-600 text-white flex items-center justify-center font-black text-[10px] shrink-0">
                          {currentUser.avatar}
                        </div>
                      )}
                      <span>Logged in: {currentUser.name} (Sign Out)</span>
                    </>
                  ) : (
                    <>
                      <svg className="w-4 h-4 shrink-0" viewBox="0 0 24 24">
                        <path fill="#4285F4" d="M22.56 12.25c0-.78-.07-1.53-.2-2.25H12v4.26h5.92c-.26 1.37-1.04 2.53-2.21 3.31v2.77h3.57c2.08-1.92 3.28-4.74 3.28-8.09z" />
                        <path fill="#34A853" d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z" />
                        <path fill="#FBBC05" d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.06H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.94l2.85-2.22.81-.63z" />
                        <path fill="#EA4335" d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.06l3.66 2.84c.87-2.6 3.3-4.52 6.16-4.52z" />
                      </svg>
                      <span>Continue with Google</span>
                    </>
                  )}
                </button>
              </div>

              {/* Mobile 2x2 Features Grid */}
              <div className="grid grid-cols-2 gap-2 mt-4 lg:hidden w-full max-w-md">
                {/* Feature 1: Offline Access */}
                <FeatureSpotlight featureId="offline-download-v1" label="Download all chapters offline!" side="top" color="blue">
                  <button
                    onClick={handleOfflineFeatureClick}
                    className={`flex items-center gap-2 p-2 border rounded-xl text-left hover:scale-[1.02] active:scale-[0.98] transition-all cursor-pointer w-full ${
                      isAlreadyDownloaded
                        ? "bg-emerald-500/10 dark:bg-emerald-900/20 border-emerald-500/30"
                        : "bg-slate-100/40 dark:bg-slate-900/40 border-slate-200/50 dark:border-slate-700/50 hover:border-blue-500/30"
                    }`}
                  >
                    <div className={`p-1.5 rounded-lg shrink-0 ${
                      isAlreadyDownloaded ? "bg-emerald-500/15 text-emerald-500" : "bg-blue-500/10 text-blue-500"
                    }`}>
